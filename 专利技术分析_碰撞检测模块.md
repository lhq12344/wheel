### 2.2 碰撞检测算法

**核心方法**: [`TryFindCollision()`](Assets/ArmCollisionMonitor.cs:256)

```csharp
private bool TryFindCollision(out string collisionMessage)
{
    collisionMessage = string.Empty;

    // 遍历所有机械臂Collider
    foreach (Collider armCollider in _armColliders)
    {
        // 遍历所有禁止碰撞体
        foreach (Collider forbiddenCollider in _forbiddenColliders)
        {
            // 1. 边界盒预检（快速排除）
            if (!armCollider.bounds.Intersects(forbiddenCollider.bounds))
            {
                continue;
            }

            // 2. 精确穿透深度计算
            bool overlapping = Physics.ComputePenetration(
                armCollider, armCollider.transform.position, armCollider.transform.rotation,
                forbiddenCollider, forbiddenCollider.transform.position, forbiddenCollider.transform.rotation,
                out Vector3 direction, out float distance);

            // 3. 有效穿透阈值判断
            float effectivePenetrationThreshold = GetEffectivePenetrationThreshold(
                armCollider, forbiddenCollider);

            if (!overlapping || distance <= effectivePenetrationThreshold)
            {
                continue;  // 可接受的微小穿透
            }

            // 4. 碰撞发生
            collisionMessage = $"Arm collider '{armCollider.transform.name}' " +
                $"collided with '{forbiddenCollider.transform.name}' " +
                $"(penetration {distance:F4}m).";
            return true;
        }
    }

    return false;
}
```

## 三、关节空间运动碰撞检测

### 3.1 评估方法

```csharp
public bool EvaluateMotionCollision(
    float[] startAnglesDeg,
    float[] targetAnglesDeg,
    out ArmCollisionGuardResult result)
{
    result = new ArmCollisionGuardResult { allowed = true };

    // 1. 计算关节角度最大差值
    float maxDelta = 0f;
    for (int i = 0; i < 6; i++)
    {
        maxDelta = Mathf.Max(maxDelta, Mathf.Abs(targetAnglesDeg[i] - startAnglesDeg[i]));
    }

    // 2. 确定采样数量
    int sampleCount = Mathf.Clamp(
        Mathf.CeilToInt(maxDelta / Mathf.Max(0.5f, _collisionMonitor.previewStepDegrees)),
        1,
        Mathf.Max(1, _collisionMonitor.maxPreviewSamples));

    // 3. 沿轨迹采样检测
    float[] sampleAngles = new float[6];
    for (int sampleIndex = 0; sampleIndex <= sampleCount; sampleIndex++)
    {
        float t = sampleCount <= 0 ? 1f : sampleIndex / (float)sampleCount;

        // 插值计算采样点关节角度
        for (int i = 0; i < 6; i++)
        {
            sampleAngles[i] = Mathf.Lerp(startAnglesDeg[i], targetAnglesDeg[i], t);
        }

        // 计算该采样点的连杆世界位姿
        Pose[] linkPoses = ComputeLinkPosesWorld(sampleAngles);

        // 预测碰撞检测
        if (_collisionMonitor.EvaluatePredictedCollision(linkPoses, out result))
        {
            result.sampleIndex = sampleIndex;
            return true;
        }
    }

    return false;
}
```

```csharp
public bool IsBasePoseCollisionFree(
    Vector3 position,
    float radius,
    IReadOnlyList<Collider> obstacles,
    out Collider hitCollider)
{
    hitCollider = null;

    // 计算胶囊体端点
    Vector3 bottom = position;
    bottom.y = 0f;
    Vector3 top = bottom + Vector3.up * Mathf.Max(0.05f, (halfHeight * 2f) - (radius * 2f));

    // 使用OverlapCapsuleNonAlloc进行高效检测
    int hitCount = Physics.OverlapCapsuleNonAlloc(
        bottom, top,
        Mathf.Max(0.05f, radius),
        _overlapBuffer,
        ~0,  // 所有层
        QueryTriggerInteraction.Ignore);

    // 过滤掉机器人自身Collider
    for (int i = 0; i < hitCount; i++)
    {
        Collider hit = _overlapBuffer[i];

        // 跳过非障碍物
        if (obstacles == null || !obstacles.Contains(hit))
        {
            continue;
        }

        hitCollider = hit;
        return false;  // 有碰撞
    }

    return true;  // 无碰撞
}
```

### 4.3 路径段碰撞检测

**文件**: [`Assets/PlannerPhysicsQueries.cs:344-368`](Assets/PlannerPhysicsQueries.cs:344)

```csharp
public bool IsSegmentCollisionFree(
    Vector3 start,
    Vector3 end,
    float radius,
    IReadOnlyList<Collider> obstacles,
    out Collider hitCollider,
    bool allowStartOccupied = false)
{
    float distance = Vector3.Distance(start, end);
    float sampleSpacing = Mathf.Max(0.05f, radius * 0.5f);

    // 沿路径采样
    int sampleCount = Mathf.Max(1, Mathf.CeilToInt(distance / sampleSpacing));

    bool escapedInitialOverlap = !allowStartOccupied;
    bool sawFreePose = false;

    for (int i = 0; i <= sampleCount; i++)
    {
        float t = i / (float)sampleCount;
        Vector3 point = Vector3.Lerp(start, end, t);

        if (!IsBasePoseCollisionFree(point, radius, obstacles, out hitCollider))
        {
            // 起点被允许占用时，只在离开起点后的碰撞才判定为失败
            if (allowStartOccupied && !escapedInitialOverlap && i < sampleCount)
            {
                // 还在起点附近，继续检查
            }
            else
            {
                return false;  // 路径有碰撞
            }
        }
        else
        {
            sawFreePose = true;
            escapedInitialOverlap = true;
        }
    }

    return sawFreePose;
}
```

```csharp
    Vector3 shadowPosition = _shadowController.rb.position;

private bool HasColliderPenetration(out string collisionMessage)
{
    // 遍历影子底盘的所有Collider
    for (int shadowIndex = 0; shadowIndex < _shadowBaseColliders.Count; shadowIndex++)
    {
        for (int obstacleIndex = 0; obstacleIndex < _activeObstacles.Count; obstacleIndex++)
        {
            // Physics.ComputePenetration精确计算
            if (!Physics.ComputePenetration(
                shadowCollider, shadowCollider.transform.position, shadowCollider.transform.rotation,
                obstacle, obstacle.transform.position, obstacle.transform.rotation,
                out Vector3 direction, out float distance))
            {
                continue;
            }

            // 穿透阈值判断
            if (distance <= 1e-4f)
            {
                continue;
            }

            collisionMessage = $"Shadow base collided with obstacle...";
            return true;
        }
    }
}
```

```csharp
private bool TryDetectShadowArmCollision(out string collisionMessage)
{
    collisionMessage = string.Empty;

    // 遍历影子机械臂的所有Collider
    for (int shadowIndex = 0; shadowIndex < _shadowArmColliders.Count; shadowIndex++)
    {
        Collider shadowCollider = _shadowArmColliders[shadowIndex];

        for (int obstacleIndex = 0; obstacleIndex < _activeObstacles.Count; obstacleIndex++)
        {
            Collider obstacle = _activeObstacles[obstacleIndex];

            // 边界盒预检
            if (!shadowCollider.bounds.Intersects(obstacle.bounds))
            {
                continue;
            }

            // 精确穿透计算
            if (!Physics.ComputePenetration(
                shadowCollider, shadowCollider.transform.position, shadowCollider.transform.rotation,
                obstacle, obstacle.transform.position, obstacle.transform.rotation,
                out Vector3 direction, out float distance))
            {
                continue;
            }

            // 获取有效穿透阈值
            float effectiveThreshold = GetEffectivePenetrationThreshold(shadowCollider, obstacle);

            if (distance <= effectiveThreshold)
            {
                continue;
            }

            collisionMessage = $"Shadow arm collided with obstacle '{obstacle.name}' " +
                $"via '{shadowCollider.name}' at (...).";
            return true;
        }
    }

    return false;
}
```

```csharp
public SafetyGateDecision EvaluateBaseCommand(
    MirrorSnapshot snapshot,
    BaseGateCommand command,
    SafetyGateRuntimeContext context)
{
    SafetyGateDecision decision = CreateDefaultGateDecision(...);

    // 1. 沿轨迹采样
    int previewSampleCount = EstimateBasePreviewSampleCount(
        command.fromWorldPosition,
        command.toWorldPosition,
        basePreviewSampleSpacingMeters,  // 0.08m
        decision.predictorState.predictedLeadTimeSeconds);

    bool collisionDetected = false;
    Collider blockingCollider = null;

    for (int sampleIndex = 0; sampleIndex <= previewSampleCount; sampleIndex++)
    {
        float t = sampleIndex / (float)previewSampleCount;
        Vector3 samplePosition = Vector3.Lerp(command.fromWorldPosition, command.toWorldPosition, t);

        // 重叠检测
        if (TryOverlapObstacle(samplePosition, inflatedBaseRadius, context.obstacles, context.obstacleLookup, out Collider sampleHitCollider))
        {
            collisionDetected = true;
            blockingCollider = sampleHitCollider;
            break;
        }

        // 净空计算
        float sampleClearance = ComputeSphereClearance(samplePosition, inflatedBaseRadius, context.obstacles);
        if (sampleClearance < previewWorstClearance)
        {
            previewWorstClearance = sampleClearance;
        }
    }

    // 2. 决策
    if (collisionDetected)
    {
        decision.type = SafetyGateDecisionType.Block;
        decision.blockCategory = SafetyGateBlockCategory.GateBlockedByCollision;
        return decision;
    }

    // 3. 净空阈值判断
    float blockClearance = Mathf.Max(0.001f, context.gateBlockClearanceMeters);
    float throttleClearance = Mathf.Max(blockClearance, context.gateThrottleClearanceMeters);

    if (minClearance <= blockClearance)
    {
        decision.type = SafetyGateDecisionType.Block;
        decision.blockCategory = SafetyGateBlockCategory.GateBlockedByClearance;
    }
    else if (minClearance <= throttleClearance)
    {
        decision.type = SafetyGateDecisionType.Throttle;
        decision.throttleRatio = ComputeThrottleRatio(minClearance, blockClearance, throttleClearance);
    }

    return decision;
}
```

```csharp
    MirrorSnapshot snapshot,
    ArmGateCommand command,
    SafetyGateRuntimeContext context,
    Arm6DOFFKController armController)

            decision.blockCategory = SafetyGateBlockCategory.GateBlockedBySingularity;
```

## 七、关键算法总结

### 7.1 穿透深度计算

**Physics.ComputePenetration** 是Unity提供的精确碰撞穿透计算API：

```csharp
bool overlapping = Physics.ComputePenetration(
    collider1,                     // 第一个碰撞体
    position1,                      // 第一个碰撞体世界位置
    rotation1,                      // 第一个碰撞体世界旋转
    collider2,                     // 第二个碰撞体
    position2,                      // 第二个碰撞体世界位置
    rotation2,                      // 第二个碰撞体世界旋转
    out Vector3 direction,          // 穿透方向（从collider2指向collider1）
    out float distance              // 穿透深度
);
```

### 7.2 有效穿透阈值

```csharp
private float GetEffectivePenetrationThreshold(Collider primary, Collider secondary)
{
    // 基于contactOffset计算有效阈值
    return Mathf.Max(0.0001f,
        GetSafeContactOffset(primary) + GetSafeContactOffset(secondary));
}

private float GetSafeContactOffset(Collider collider)
{
    return collider != null ? Mathf.Max(0f, collider.contactOffset) : 0f;
}
```

### 7.3 性能优化技术

| 技术                  | 说明                                                |
| --------------------- | --------------------------------------------------- |
| OverlapSphereNonAlloc | 预分配缓冲区，避免每帧GC                            |
| Bounds.Intersects     | 快速排除明显不相交的碰撞体                          |
| 分层检测              | 先用Overlap快速筛选，再用ComputePenetration精确计算 |
| 采样策略              | 根据轨迹长度自适应调整采样密度                      |
