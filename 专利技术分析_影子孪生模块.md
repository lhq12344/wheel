## 一、系统概述

影子孪生系统是预测性安全防护的核心，它在虚拟空间中创建一个与真实机器人物理隔离的副本（影子机器人），通过让影子机器人"超前执行"命令来验证安全性，从而在危险发生前阻止不安全的命令下发到真实机器人。

```csharp
private bool TryDetectShadowBaseCollision(out string collisionMessage)
{
    // 1. 基础碰撞检测
    if (_physicsQueries.IsBasePoseCollisionFree(shadowPosition, _activeBaseRadius, _activeObstacles, out _))
    {
        return false;
    }

    // 2. 精确穿透深度计算
    if (!Physics.ComputePenetration(
        shadowCollider, shadowCollider.transform.position, shadowCollider.transform.rotation,
        obstacle, obstacle.transform.position, obstacle.transform.rotation,
        out Vector3 direction, out float distance))
    {
        continue;
    }

    // 3. 阈值判断
    if (distance <= 1e-4f)
    {
        continue;  // 可接受的微小穿透
    }
}
```

```csharp
private bool TryDetectShadowArmCollision(out string collisionMessage)
{
    for (int shadowIndex = 0; shadowIndex < _shadowArmColliders.Count; shadowIndex++)
    {
        Collider shadowCollider = _shadowArmColliders[shadowIndex];

        for (int obstacleIndex = 0; obstacleIndex < _activeObstacles.Count; obstacleIndex++)
        {
            Collider obstacle = _activeObstacles[obstacleIndex];

            // 边界盒预检
            if (!shadowCollider.bounds.Intersects(obstacle.bounds))
            {
                continue;
            }

            // 精确穿透计算
            if (!Physics.ComputePenetration(...))
            {
                continue;
            }

            // 判断穿透深度是否超过阈值
            if (distance <= GetEffectivePenetrationThreshold(...))
            {
                continue;
            }

            // 碰撞发生
            collisionMessage = $"Shadow arm collided with obstacle '{obstacle.name}'...";
            return true;
        }
    }
}
```

### 4.4 预测位姿积分

**代码位置**: [`ShadowRobotVisualizer.cs:1050-1090`](Assets/ShadowRobotVisualizer.cs:1050)

```csharp
private static void IntegrateBasePose(
    ref Vector3 worldPosition,
    ref Quaternion worldRotation,
    float linearVelocity,
    float angularVelocity,
    float deltaTime)
{
    Vector3 planarForward = planarRotation * Vector3.forward;

    if (Mathf.Abs(angularVelocity) <= 1e-4f)
    {
        // 直线运动
        worldPosition = startPosition + (planarForward * linearVelocity * deltaTime);
    }
    else
    {
        // 圆弧运动
        float deltaYawRad = angularVelocity * deltaTime;
        float radius = linearVelocity / angularVelocity;

        Vector3 planarDisplacement =
            (planarForward * (radius * Mathf.Sin(deltaYawRad)))
            + (planarRight * (radius * (1f - Mathf.Cos(deltaYawRad))));

        worldPosition = startPosition + planarDisplacement;
    }
}
```

```csharp
private static int EstimateBasePreviewSampleCount(
    Vector3 fromWorldPosition,
    Vector3 toWorldPosition,
    float spacingMeters,
    float lookaheadSeconds)
{
    float planarDistance = Vector3.Distance(fromWorldPosition, toWorldPosition);
    float effectiveSpacing = Mathf.Max(0.02f, spacingMeters);

    int distanceSamples = Mathf.CeilToInt(planarDistance / effectiveSpacing);
    int leadSamples = Mathf.CeilToInt(Mathf.Max(MinLookaheadFloorSeconds, lookaheadSeconds) / Time.fixedDeltaTime);

    return Mathf.Clamp(Mathf.Max(distanceSamples, leadSamples), 2, 64);
}
```

### 5.4 奇异性检测

**代码位置**: [`ShadowSimulationGate.cs:900-927`](Assets/ShadowSimulationGate.cs:900)

```csharp
private static float ComputeLinearJacobianGramDeterminant(float[,] jacobian)
{
    // 提取线性雅可比部分 (3x6)
    float[,] jjt = new float[3, 3];
    for (int row = 0; row < 3; row++)
    {
        for (int column = 0; column < 3; column++)
        {
            float sum = 0f;
            for (int jointIndex = 0; jointIndex < 6; jointIndex++)
            {
                sum += jacobian[3 + row, jointIndex] * jacobian[3 + column, jointIndex];
            }
            jjt[row, column] = sum;
        }
    }

    // 计算3x3矩阵行列式
    float determinant =
        (jjt[0, 0] * ((jjt[1, 1] * jjt[2, 2]) - (jjt[1, 2] * jjt[2, 1])))
        - (jjt[0, 1] * ((jjt[1, 0] * jjt[2, 2]) - (jjt[1, 2] * jjt[2, 0])))
        + (jjt[0, 2] * ((jjt[1, 0] * jjt[2, 1]) - (jjt[1, 1] * jjt[2, 0])));

    return Mathf.Max(0f, determinant);
}
```

```csharp
private static float ComputeThrottleRatio(
    float minClearance,
    float blockThreshold,
    float throttleThreshold)
{
    // 线性插值计算限速比例
    float normalized = Mathf.InverseLerp(blockThreshold, throttleThreshold, minClearance);
    return Mathf.Clamp(Mathf.Lerp(0.25f, 0.85f, normalized), 0.2f, 0.9f);
}
```

### 5.6 超前时间计算

**代码位置**: [`ShadowSimulationGate.cs:929-954`](Assets/ShadowSimulationGate.cs:929)

```csharp
private static float ComputeLookaheadSeconds(SafetyGateRuntimeContext context, MirrorSnapshot snapshot, bool armMode)
{
    // 基础窗口：通信延迟 + 计算预算
    float baseWindow = Mathf.Max(
        context.minLookaheadSeconds,
        context.commLatencySeconds + context.computeBudgetSeconds);

    if (!armMode)
    {
        // 底盘：取制动距离对应时间
        float baseBrake = snapshot.basePlanarSpeed / context.baseBrakeDecelMetersPerSecond2;
        return Mathf.Max(baseWindow, baseBrake);
    }
    else
    {
        // 机械臂：取最大关节速度对应制动时间
        float maxJointSpeed = GetMaxJointSpeed(snapshot.armJointVelocitiesDegPerSecond);
        float armBrake = maxJointSpeed / context.armBrakeDecelDegPerSecond2;
        return Mathf.Max(baseWindow, armBrake);
    }
}
```
